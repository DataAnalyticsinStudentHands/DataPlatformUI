// types/projectTypes.js (revised)

export function createEmptyProject(userId = '') {
  return {
    id: generateId(),
    
    // === MANDATORY DATA ===
    label: { icon: 'layers', text: '' },
    title: '',
    description: '',
    authors: [createEmptyAuthor()],  // Always array, min 1
    tags: [],
    findings: [
      createEmptyFinding({ variant: 'data' }),
    ],
    conclusion: { text: '', attribution: '' },
    footer: createDefaultFooter(),
    
    // === OPTIONAL SECTIONS (enabled via config) ===
    partners: [],
    milestones: [],      // Timeline
    impactItems: [],
    poster: null,        // null = disabled, object = enabled
    
    // === SECTION CONFIGURATION ===
    enabledSections: ['partners'],  // IDs of optional sections user enabled
    
    // === METADATA ===
    metadata: createMetadata(userId),
  };
}

      :label="project.label"

          v-if="isSectionEnabled('timeline')"
          :milestones="project.milestones"
        />

          v-if="isSectionEnabled('impact')"
          :impact-items="project.impactItems"
        />
      </div>

      <!-- MAIN ZONE -->
      <div class="main">
        <!-- Optional: Poster (order: 60) -->
        <template v-if="isSectionEnabled('poster')">
          <PosterCardPdf v-if="project.poster?.type === 'pdf'" ... />
          <PosterCardImage v-else ... />
        </template>
        
        <!-- Empty state if no poster -->
        <div v-else class="no-poster-placeholder">
          <!-- Optional: could show something here -->
        </div>
      </div>
    </div>

    <!-- FOOTER ZONE (always rendered) -->
    <FooterBanner :footer="project.footer" />
  </div>
</template>

      <FindingsInput v-model="formData.findings" />
      <ConclusionInput v-model="formData.conclusion" />
    </FormSection>

    <!-- OPTIONAL SECTIONS (rendered based on enabledSections) -->
    <template v-for="section in sortedOptionalSections" :key="section.id">
      <FormSection 
        :number="getFormSectionNumber(section)"
        :title="section.name"
        :removable="true"
        @remove="disableSection(section.id)"
      >
        <component :is="getSectionFormComponent(section.id)" v-model="formData" />
      </FormSection>
    </template>

    <!-- ADD SECTION BUTTON -->
    <SectionAddMenu
      :available-sections="availableOptionalSections"
      @add="enableSection"
    />
  </v-form>
</template>
8. Section Order Enforcement
Sorting Logic

// Always render sections in this order, regardless of when user enabled them:

const SECTION_ORDER = [
  // Mandatory
  'hero',      // 0
  'authors',   // 1
  'tags',      // 10
  'findings',  // 20
  
  // Optional (sorted by order value)
  'partners',  // 30
  'timeline',  // 40
  'impact',    // 50
  'poster',    // 60
  
  // Mandatory
  'footer',    // 100
];

// When rendering:
const sortedEnabledSections = computed(() => {
  return OPTIONAL_SECTIONS
    .filter(s => enabledSections.includes(s.id))
    .sort((a, b) => a.order - b.order);
});

10. Data Migration
For any existing drafts in localStorage:


function migrateProjectData(oldProject) {
  // Detect old format
  if (oldProject.templateType) {
    const wasResearch = oldProject.templateType === 'research';
    
    return {
      ...oldProject,
      // Convert single author to array
      authors: wasResearch 
        ? [oldProject.author] 
        : oldProject.authors,
      // Determine enabled sections based on data presence
      enabledSections: [
        oldProject.partners?.length > 0 && 'partners',
        oldProject.milestones?.length > 0 && 'timeline',
        oldProject.impactItems?.length > 0 && 'impact',
        oldProject.poster?.url && 'poster',
      ].filter(Boolean),
      // Remove old fields
      templateType: undefined,
      author: undefined,
    };
  }
  return oldProject;
}
